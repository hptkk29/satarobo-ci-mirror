function mulberry32(a: number) { return () => { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const rnd = mulberry32(20261101);
const ri = (lo: number, hi: number) => lo + Math.floor(rnd() * (hi - lo + 1));

const MA = ["MN", "TH", "THCS", "PH", "QS"] as const;
type Case = { ten: string; boDe: string; giay: number | null; traLoi: Record<string, unknown>; goc: unknown };
const cases: Case[] = [];

  const bo = napBoDe(boDe);
  const goc = cham(bo, traLoi as Record<string, number>, giay);
  cases.push({ ten, boDe, giay, traLoi, goc: JSON.parse(JSON.stringify(goc)) });
}

  // Biên
  for (let v = 1; v <= max; v++) them(`${ma}/toan-${v}`, ma, full(() => v));

  // Thiếu câu: bỏ 1, bỏ nửa, rỗng
  const dayDu = full(() => ri(1, max));
  const bo1 = { ...dayDu }; delete bo1[cau[0].ma]; them(`${ma}/thieu-1`, ma, bo1);
  const boNua = Object.fromEntries(Object.entries(dayDu).filter((_e, i) => i % 2 === 0)); them(`${ma}/thieu-nua`, ma, boNua);

    const t = { ...dayDu }; t[cau[1].ma] = v as unknown as number; them(`${ma}/la-${nhan}`, ma, t);
  }
  // Vùng lệch: mọi câu ở mức giữa trừ vài câu (kiểm ngưỡng phẳng 0.4)
  if (max % 2 === 1) { const g = (max + 1) / 2; for (const nPhang of [Math.floor(cau.length * 0.35), Math.ceil(cau.length * 0.45)]) them(`${ma}/phang-${nPhang}`, ma, full((_c, i) => (i < nPhang ? g : (i % 2 ? max : 1)))); }
}
